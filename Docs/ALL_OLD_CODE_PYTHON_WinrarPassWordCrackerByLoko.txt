import os
import subprocess
import itertools
import string
import sys
import time

# ============ CONFIG ============
RAR_FILE = "Docs/rar_examples/abc.rar"
UNRAR = "UnRAR.exe"

# Charset a usar (personalízalo)
CHARSET = string.ascii_lowercase  # a-z
# CHARSET = string.ascii_lowercase + string.digits          # a-z 0-9
# CHARSET = string.ascii_letters + string.digits            # a-z A-Z 0-9
# CHARSET = string.ascii_letters + string.digits + "._-@:/" # + símbolos

MIN_LEN = 1
MAX_LEN = 4   # 👈 cuidado: cada carácter más multiplica el tiempo
# =======================================


def get_paths():
    base = os.path.dirname(os.path.abspath(__file__))
    return os.path.join(base, RAR_FILE), os.path.join(base, UNRAR)


def try_password(rar_path, unrar_path, password):
    """Devuelve True si la contraseña es correcta."""
    try:
        result = subprocess.run(
            [unrar_path, 't', '-p' + password, rar_path],
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
            timeout=10
        )
        return result.returncode == 0
    except Exception:
        return False


def brute_force(charset, min_len, max_len):
    """Generador que produce todas las combinaciones posibles."""
    for length in range(min_len, max_len + 1):
        for combo in itertools.product(charset, repeat=length):
            yield ''.join(combo)


def extract_content(rar_path, unrar_path, password):
    """Extrae el RAR a una carpeta y muestra el contenido."""
    out_dir = os.path.join(os.path.dirname(rar_path), "extraido")
    os.makedirs(out_dir, exist_ok=True)

    print(f"\n📦 Extrayendo con contraseña '{password}'...")
    result = subprocess.run(
        [unrar_path, 'x', '-p' + password, '-o+', rar_path, out_dir + os.sep],
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True
    )

    if result.returncode != 0:
        print("⚠️ No se pudo extraer:", result.stderr)
        return

    print(f"✅ Extraído en: {out_dir}\n")
    print("📄 Contenido de los archivos de texto encontrados:\n")

    for root, _, files in os.walk(out_dir):
        for fname in files:
            fpath = os.path.join(root, fname)
            print(f"--- {fname} ---")
            try:
                with open(fpath, 'r', encoding='utf-8', errors='ignore') as f:
                    print(f.read())
            except Exception as e:
                print(f"(No es texto legible: {e})")
            print()


def main():
    rar_path, unrar_path = get_paths()

    if not os.path.exists(rar_path):
        print(f"❌ No se encontró {RAR_FILE} en la carpeta.")
        sys.exit(1)
    if not os.path.exists(unrar_path):
        print(f"❌ No se encontró {UNRAR} en la carpeta.")
        print("   Cópialo desde C:\\Program Files\\WinRAR\\UnRAR.exe")
        sys.exit(1)

    total = sum(len(CHARSET) ** n for n in range(MIN_LEN, MAX_LEN + 1))
    print(f"🔍 Iniciando fuerza bruta")
    print(f"   Charset: {CHARSET}")
    print(f"   Longitud: {MIN_LEN} a {MAX_LEN}")
    print(f"   Combinaciones totales: {total:,}")
    print(f"   (Ctrl+C para detener)\n")

    start = time.time()
    count = 0

    try:
        for password in brute_force(CHARSET, MIN_LEN, MAX_LEN):
            count += 1
            if count % 100 == 0 or count == 1:
                elapsed = time.time() - start
                speed = count / elapsed if elapsed > 0 else 0
                print(f"   [{count}/{total}] probando: {password}  ({speed:.1f}/s)", end='\r')

            if try_password(rar_path, unrar_path, password):
                elapsed = time.time() - start
                print(f"\n\n🎉 ¡CONTRASEÑA ENCONTRADA!: '{password}'")
                print(f"   Probadas {count} combinaciones en {elapsed:.1f} segundos.\n")
                extract_content(rar_path, unrar_path, password)
                return

    except KeyboardInterrupt:
        print("\n\n⏹️ Detenido por el usuario.")
        return

    print("\n\n❌ No se encontró la contraseña en el rango probado.")


if __name__ == "__main__":
    main()

"""
FelipedelosH
2026

Main application controller responsible for orchestrating
brute-force strategies, initializing required resources,
managing dependencies, and saving execution logs.
"""
import os
from datetime import datetime

from src.CORE.Config import Config
from src.CORE.FolderController import FolderController
from src.Infrastructure.services.WinrarService import WinrarService
from src.Models.Enums.Attack import Attack
from src.Models.Entities.WorkState import WorkState
from src.Models.Enums.WorkStatus import WorkStatus

class Controller:
    def __init__(self, path: str, config: Config, dependencies: dict):
        self.path = path
        self.dependencies = dependencies
        self.output_path = os.path.join(self.path, "OUTPUT")
        self.currentDate = datetime.now().strftime("%Y-%m-%d %H:%M")
        self.config = config
        self.logger = dependencies["logger"]
        self.logger.info(f"Controller::__init__::START_AT::{self.currentDate}")
        self.CHARSET_FILE = self.config.getConfig("GEN_CONFIG_CHARACTERS_ARR_INPUT_FILE")
        self.MIN_LEN = self.config.getConfig("GEN_CONFIG_LEN_MIN_SIZE_CHAR_SELECT_TO_CRACK")
        self.MAX_LEN = self.config.getConfig("GEN_CONFIG_LEN_MAX_SIZE_CHAR_SELECT_TO_CRACK")
        self.RAR_FILE = self.config.getConfig("RAR_FILE_TO_CRACK")
        self.rar_path = os.path.join(self.path, self.RAR_FILE)
        self.unrar_path = os.path.join(self.path, self.config.getConfig("UNRAR_SOFTWARE_FILE_PATH"))
        FolderController.createInitalFolders(self.path, self.logger)

        # Vars
        self.work_state = None

        self.loadPreviousWork()


    def loadPreviousWork(self):
        self.logger.info("Controller::loadPreviousWork::TRY")

        repository = self.dependencies["work_repository"]

        # RAR must exist before we can compute its identity hash
        if not os.path.isfile(self.rar_path):
            raise FileNotFoundError(f"RAR file not found: {self.rar_path}")

        # Compute current identity
        charset = self._loadCharset()
        min_len = int(self.MIN_LEN)
        max_len = int(self.MAX_LEN)

        if min_len < 1 or max_len < min_len:
            raise ValueError("Invalid password length range.")

        attack_mode = self.config.getConfig("GEN_CONFIG_ATTACK_MODE")
        rar_sha256 = WinrarService.computeSha256(self.rar_path)
        rar_file = os.path.relpath(self.rar_path, self.path)
        rar_size_bytes = os.path.getsize(self.rar_path)
        total_combinations = sum(
            len(charset) ** length
            for length in range(min_len, max_len + 1)
        )

        # LOAD PREVIOUS WORK
        if repository.exists():
            work = None
            try:
                work = repository.load()
            except (OSError, ValueError, KeyError, TypeError) as e:
                # Corrupt or incompatible version: back up (preserve) and create new
                self.logger.error(f"Controller::loadPreviousWork::CORRUPT::{e}")
                try:
                    backup_path = repository.backupCorrupt()
                    self.logger.info(f"Controller::loadPreviousWork::BACKED_UP::{backup_path}")
                except Exception as be:
                    self.logger.error(f"Controller::loadPreviousWork::BACKUP_FAILED::{be}")
                work = None

            if work is not None:
                if self._isCompatibleWork(work, charset, min_len, max_len, attack_mode, rar_sha256):
                    self.work_state = work
                    self.logger.info(
                        f"Controller::loadPreviousWork::FOUND::"
                        f"STATUS::{self.work_state.status}"
                    )
                    self.logger.info(
                        f"Controller::loadPreviousWork::RESUME::"
                        f"CURRENT_COLONY::{self.work_state.current_colony}::"
                        f"CURRENT_INDEX::{self.work_state.current_colony_index}"
                    )
                    return self.work_state

                # Incompatible identity: archive previous (preserve) and create new
                self.logger.warn("Controller::loadPreviousWork::IDENTITY_MISMATCH")
                try:
                    archive_path = repository.archive()
                    self.logger.info(f"Controller::loadPreviousWork::ARCHIVED::{archive_path}")
                except Exception as ae:
                    self.logger.error(f"Controller::loadPreviousWork::ARCHIVE_FAILED::{ae}")

        # CREATE NEW WORK
        self.logger.info("Controller::loadPreviousWork::NOT_FOUND_OR_ARCHIVED")

        self.work_state = WorkState(
            rar_file=rar_file,
            rar_sha256=rar_sha256,
            rar_size_bytes=rar_size_bytes,
            charset=charset,
            min_len=min_len,
            max_len=max_len,
            attack_mode=attack_mode,
            status=WorkStatus.NEW,
            password_found=False,
            total_colonies=max_len - min_len + 1,
            total_combinations=total_combinations,
            created_at=datetime.now().isoformat(),
            updated_at=datetime.now().isoformat()
        )

        repository.save(self.work_state)

        self.logger.info(
            "Controller::loadPreviousWork::CREATE::NEW"
        )

        return self.work_state

    def _loadCharset(self) -> str:
        charset_path = os.path.join(self.path, self.CHARSET_FILE)

        with open(charset_path, "r", encoding="utf-8") as file:
            content = file.read()

        charset = "".join(c for c in content if c not in ", \n\r\t")
        charset = "".join(dict.fromkeys(charset))

        if not charset:
            raise ValueError("The charset cannot be empty.")

        return charset

    def _isCompatibleWork(self, work, charset: str, min_len: int, max_len: int, attack_mode: str, rar_sha256: str) -> bool:
        """A previous work is reusable only if charset, lengths, strategy and RAR
        content hash all match. An empty stored hash is NOT a validated identity."""
        if work.charset != charset:
            return False
        if work.min_len != min_len or work.max_len != max_len:
            return False
        if work.attack_mode != attack_mode:
            return False
        if not work.rar_sha256:
            return False
        if work.rar_sha256 != rar_sha256:
            return False
        return True


    def bruteForce(self):
        _mode = self.config.getConfig("GEN_CONFIG_ATTACK_MODE")
        self.logger.info(f"Controller::bruteForce::TRY::{_mode}")
        if _mode == str(Attack.RANDOM):
            pass

        if _mode == str(Attack.SEQUENTIAL):
            return self.dependencies["sequential_brute_force"].execute()
        
        return None

    def extractAndPrint(self, password):
        self.logger.info(f"Controller::extractAndPrint::TRY::{password}")
        out_dir = WinrarService.extractContent(self.rar_path, self.unrar_path, password, self.output_path)
        if not out_dir:
            self.logger.error(f"Controller::extractAndPrint::ERR_PATH::{out_dir}::NOT_FOUND")
            return

        self.logger.info(f"Controller::extractAndPrint::EXTRACT::{out_dir}")
        for root, _, files in os.walk(out_dir):
            for fname in files:
                if ".log" in fname:
                    continue
                fpath = os.path.join(root, fname)
                size = os.path.getsize(fpath)
                self.logger.info(f"Controller::extractAndPrint::FILE::{fname}::{size}_bytes")

    def saveLogs(self):
        try:
            timestamp = datetime.now().strftime("%Y_%m_%d_%H_%M")
            filename = f"logs_{timestamp}.log"
            full_path = os.path.join(self.output_path, filename)

            with open(full_path, "w", encoding="UTF-8") as f:
                f.write(self.logger.log)
            print(f"LOGs: {full_path}")
        except Exception as e:
            print(f"ERROR FATAL IN saveLogs: {e}")

        unrar_path = config.getConfig("UNRAR_SOFTWARE_FILE_PATH")

            work_path=work_path

            content = file.read()

        charset = "".join(
            c for c in content if c not in ", \n\r\t"
        )

        if not charset:
            raise ValueError("The charset cannot be empty.")

            temp_path=temp_path,
            output_path=output_path,
            work_repository=work_repository,
            logger=logger
        )

            "work_repository": work_repository,
            "sequential_brute_force": sequential_brute_force,
            "logger": logger
        }

# src\Infrastructure\repositories\JsonWorkRepository.py
import os
import json
from dataclasses import asdict
from datetime import datetime
from typing import Optional
from src.Application.repositories.IWorkRepository import IWorkRepository
from src.Models.Entities.WorkState import WorkState
from src.Models.Enums.WorkStatus import WorkStatus


class JsonWorkRepository(IWorkRepository):
    def __init__(self, work_path: str):
        self.work_path = work_path

    def exists(self) -> bool:
        return os.path.isfile(self.work_path)

    def load(self) -> Optional[WorkState]:
        if not self.exists():
            return None

        with open(self.work_path, "r", encoding="utf-8") as file:
            data = json.load(file)

        if data.get("version") != 1:
            raise ValueError("Unsupported work.json version.")

        data["status"] = WorkStatus[data["status"]]

        return WorkState(**data)

    def save(self, state: WorkState) -> None:
        directory = os.path.dirname(os.path.abspath(self.work_path))
        os.makedirs(directory, exist_ok=True)

        data = asdict(state)
        data["status"] = state.status.name

        temp_path = self.work_path + ".tmp"

        with open(temp_path, "w", encoding="utf-8") as file:
            json.dump(data, file, indent=4)
            file.flush()
            os.fsync(file.fileno())

        os.replace(temp_path, self.work_path)

    def archive(self) -> Optional[str]:
        if not self.exists():
            return None

        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S_%f")

        directory = os.path.dirname(self.work_path)
        archive_path = os.path.join(
            directory,
            f"work.archived.{timestamp}.json"
        )

        os.replace(self.work_path, archive_path)

        return archive_path

    def backupCorrupt(self) -> Optional[str]:
        """Renombra work.json a work.corrupt.{timestamp}.json preservando el original."""
        if not self.exists():
            return None

        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S_%f")

        directory = os.path.dirname(self.work_path)
        corrupt_path = os.path.join(
            directory,
            f"work.corrupt.{timestamp}.json"
        )

        os.replace(self.work_path, corrupt_path)

        return corrupt_path

            f"{self.separator} 100.00% "
            f"{self.separator} Total: {self._fmt(elapsed)}"
            + (f" {self.separator} {extra_info}" if extra_info else "")
        )
        if self.show_bar:
            print(f"[{self._bar(100)}] 100.00%")
        print("=" * 70)

    # ---------------------------------------------------
    # HELPERS INTERNOS
    # ---------------------------------------------------
    def _bar(self, percentage):
        filled = int((percentage / 100) * self.bar_size)
        return "#" * filled + "-" * (self.bar_size - filled)

    def _fmt(self, seconds):
        if seconds < 60:
            return f"{seconds:.2f} s"
        if seconds < 3600:
            return f"{seconds / 60:.2f} min"
        return f"{seconds / 3600:.2f} h"

    def newline(self):
        """Cierra la línea inline para poder escribir el mensaje final."""
        if self.inline:
            print()

    @property
    def elapsed(self):
        """Segundos transcurridos desde start()."""
        if self.start_time is None:
            return 0
        return time.perf_counter() - self.start_time

class TermiteCheckError(Exception):
    """Raised when a termite check fails with a transient technical error.

    Progress must NOT advance: the candidate was not confirmed as incorrect,
    so on resume it has to be retried. Carries the index (never the password).
    """
    def __init__(self, index: int):
        super().__init__(f"TERMITE_CHECK_ERROR::INDEX::{index}")
        self.index = index


class SequentialBruteForceService(IBruteForceService):
    def __init__(self, rar_path, unrar_path, charset, min_len, max_len, temp_path, output_path, work_repository, logger):
        self.rar_path = rar_path
        self.unrar_path = unrar_path
        self.charset = charset
        self.min_len = min_len
        self.max_len = max_len
        self.temp_path = temp_path
        self.output_path = output_path
        self.work_repository = work_repository
        self.logger = logger


    def execute(self):
        """Execute the sequential colony workflow, resuming from work.json if possible."""
        self._validate_inputs()
        total = PasswordCombinationGenerator.countTotal(
            self.charset, self.min_len, self.max_len
        )
        self.logger.info("SequentialBruteForceService::execute::TRY")
        self.logger.info(f"SequentialBruteForceService::execute::CHARS::TOTAL::{len(self.charset)}")
        self.logger.info(f"SequentialBruteForceService::execute::CHARS::{self.charset}")
        self.logger.info(f"SequentialBruteForceService::execute::TOTAL_COMBINATIONS::{total:,}")
        self.logger.info(f"SequentialBruteForceService::execute::LEN_FROM::{self.min_len}::LEN_TO::{self.max_len}")
        print("\n=========TERMITES INIT=========")
        print("PRESS: (Ctrl+C) TO STOP\n")

        # LOAD WORK (Controller already created/verified and saved it)
        work = self.work_repository.load()
        if work is None:
            self.logger.error("SequentialBruteForceService::execute::WORK_NOT_FOUND")
            return False

        # Terminal states: do not re-execute
        if work.status in (WorkStatus.FOUND, WorkStatus.COMPLETED, WorkStatus.FAILED):
            self.logger.info(
                f"SequentialBruteForceService::execute::WORK_ALREADY_FINISHED::STATUS::{work.status}"
            )
            return False

        if work.status == WorkStatus.RUNNING:
            self.logger.warn("SequentialBruteForceService::execute::PREVIOUS_RUN_NOT_CLEAN::RESUMING")

        # Defensive: stale current_colony outside the configured range
        if work.current_colony is not None and not (work.min_len <= work.current_colony <= work.max_len):
            self.logger.warn(
                f"SequentialBruteForceService::execute::STALE_CURRENT_COLONY::{work.current_colony}::RESET"
            )
            work.current_colony = None
            work.current_colony_index = -1

        # Authoritative recompute of tried combinations on resume
        if work.status in (WorkStatus.PAUSED, WorkStatus.RUNNING):
            work.total_combinations_tried = self._computeTriedCount(work)

        # EVENT: WORK_STARTED
        work.status = WorkStatus.RUNNING
        self._persist(work)
        self.logger.info("SequentialBruteForceService::execute::WORK_STARTED")

        try:
            for length in range(work.min_len, work.max_len + 1):
                # Skip completed colonies
                if length in work.completed_colonies:
                    self.logger.info(f"SequentialBruteForceService::execute::SKIP_COLONY::{length}")
                    if work.current_colony == length:
                        work.current_colony = None
                        work.current_colony_index = -1
                    continue

                total_length = PasswordCombinationGenerator.countByLength(self.charset, length)

                start_index = 0
                if length == work.current_colony and work.current_colony_index >= 0:
                    start_index = work.current_colony_index + 1

                # Colony exhausted but never marked completed (e.g. crash before COLONY_COMPLETED)
                if start_index >= total_length:
                    self.logger.info(
                        f"SequentialBruteForceService::execute::COLONY_ALREADY_EXHAUSTED::{length}"
                    )
                    self._markColonyCompleted(length, work)
                    continue

                # EVENT: COLONY_STARTED
                work.current_colony = length
                work.current_colony_index = start_index - 1
                self._persist(work)
                self.logger.info(
                    f"SequentialBruteForceService::execute::COLONY_STARTED::LEN::{length}::FROM_INDEX::{start_index}"
                )

                if self._run_colony(length, start_index, work):
                    # EVENT: PASSWORD_FOUND
                    work.status = WorkStatus.FOUND
                    work.password_found = True
                    work.completed_at = datetime.now().isoformat()
                    self._persist(work)
                    self.logger.info("SequentialBruteForceService::execute::PASSWORD_FOUND")
                    return True

                # EVENT: COLONY_COMPLETED
                self._markColonyCompleted(length, work)
        except KeyboardInterrupt:
            self.logger.warn("SequentialBruteForceService::execute::STOP_PROCESS_BY_USER")
            # EVENT: WORK_PAUSED
            work.status = WorkStatus.PAUSED
            self._persist(work)
            self.logger.info("SequentialBruteForceService::execute::WORK_PAUSED")
            return False
        except TermiteCheckError as e:
            self.logger.error(f"SequentialBruteForceService::execute::TERMITE_CHECK_ERROR::INDEX::{e.index}")
            # EVENT: WORK_PAUSED — keep last valid checkpoint, do not advance progress
            work.status = WorkStatus.PAUSED
            self._persist(work)
            self.logger.info("SequentialBruteForceService::execute::WORK_PAUSED")
            return False
        except (FileNotFoundError, PermissionError) as e:
            self.logger.error(f"SequentialBruteForceService::execute::HARD_ERROR::{e}")
            # EVENT: WORK_FAILED — unrecoverable infrastructure error
            work.status = WorkStatus.FAILED
            work.completed_at = datetime.now().isoformat()
            self._persist(work)
            self.logger.info("SequentialBruteForceService::execute::WORK_FAILED")
            return False

        # EVENT: WORK_COMPLETED
        work.status = WorkStatus.COMPLETED
        work.completed_at = datetime.now().isoformat()
        self._persist(work)
        self.logger.info(f"SequentialBruteForceService::execute::WORK_COMPLETED::FILE::{self.rar_path}")
        self.logger.info(f"SequentialBruteForceService::execute::DONT_CRACK_PASSWORD::FILE::{self.rar_path}")
        return False

    def _validate_inputs(self):
        if not os.path.isfile(self.rar_path):
            raise FileNotFoundError(f"RAR file not found: {self.rar_path}")
        if not os.path.isfile(self.unrar_path):
            raise FileNotFoundError(f"UnRAR executable not found: {self.unrar_path}")

    def _computeTriedCount(self, work) -> int:
        tried = sum(
            PasswordCombinationGenerator.countByLength(self.charset, length)
            for length in work.completed_colonies
        )
        if work.current_colony is not None and work.current_colony_index >= 0:
            tried += work.current_colony_index + 1
        return tried

    def _persist(self, work):
        work.updated_at = datetime.now().isoformat()
        self.work_repository.save(work)

    def _run_colony(self, length, start_index, work):
        self.logger.info(f"SequentialBruteForceService::_run_colony::SPAWN_COLONY::LEN::{length}")
        print(f"\n>>> SPAWNING COLONY :: LENGTH {length} :: FROM INDEX {start_index}")
        colony_path = TermiteService.spawnColony(
            self.temp_path, self.charset, length, self.logger, start_index=start_index
        )
        total = PasswordCombinationGenerator.countByLength(self.charset, length)

        tracker = BeautifulPrinter(
            label=f"TERMITES-L{length}",
            total=total,
            inline=True,
            update_every=100,
            show_current=True,
            show_speed=True,
            show_bar=False,
            header=None,
            footer=False,
        )
        tracker.start()
        tracker.current = start_index

        # Deterministic index-based iteration (NOT file positions).
        generator = PasswordCombinationGenerator.generatorForLength(self.charset, length)
        for _ in range(start_index):
            next(generator)

        for index in range(start_index, total):
            file_path = os.path.join(colony_path, f"{index}.txt")
            password = TermiteService.readTermite(file_path)
            if password is None:
                # Defensive fallback if the physical termite is missing
                password = next(generator)
            else:
                next(generator)

            tracker.update(extra_info=f"idx::{index}", current=index + 1)

            result = WinrarService.tryPassword(self.rar_path, self.unrar_path, password)

            if result == PasswordTestResult.CORRECT:
                TermiteService.survive(file_path, self.output_path)
                elapsed = tracker.elapsed
                tracker.newline()
                self.logger.info(
                    f"SequentialBruteForceService::_run_colony::TERMITE_WINNER::{index}.txt::TIME::{elapsed:.1f}s"
                )
                print("\n\nPassword found.")
                self._extract_and_log(password)
                return True
            elif result == PasswordTestResult.INCORRECT:
                TermiteService.kill(file_path)
                # EVENT: TERMITE_PROCESSED
                work.current_colony_index = index
                work.total_combinations_tried += 1
                processed_since_start = index - start_index + 1
                if work.flush_interval > 0 and processed_since_start % work.flush_interval == 0:
                    self._persist(work)
            elif result == PasswordTestResult.ERROR:
                self.logger.error(
                    f"SequentialBruteForceService::_run_colony::TERMITE_CHECK_ERROR::INDEX::{index}"
                )
                raise TermiteCheckError(index)

        tracker.newline()
        self.logger.info(f"SequentialBruteForceService::_run_colony::COLONY_DEAD::LEN::{length}")
        return False

    def _extract_and_log(self, password):
        out_dir = WinrarService.extractContent(
            self.rar_path, self.unrar_path, password, self.output_path
        )
        if not out_dir:
            self.logger.error("SequentialBruteForceService::_extract_and_log::EXTRACTION_FAILED")
            return
        self.logger.info(f"SequentialBruteForceService::_extract_and_log::EXTRACT::{out_dir}")
        for root, _, files in os.walk(out_dir):
            for fname in files:
                if ".log" in fname:
                    continue
                fpath = os.path.join(root, fname)
                size = os.path.getsize(fpath)
                self.logger.info(f"SequentialBruteForceService::_extract_and_log::FILE::{fname}::{size}_bytes")

    def _markColonyCompleted(self, length: int, work):
        self.logger.info(
            f"SequentialBruteForceService::_markColonyCompleted::TRY::LEN::{length}"
        )

        if work is None:
            raise RuntimeError("WORK_NOT_FOUND")

        # UPDATE COMPLETED COLONIES (dedup) — mutate the shared work in place
        if length not in work.completed_colonies:
            work.completed_colonies.append(length)

        # UPDATE PROGRESS
        work.current_colony = None
        work.current_colony_index = -1
        self._persist(work)

        self.logger.info(
            f"SequentialBruteForceService::_markColonyCompleted::OK::LEN::{length}"
        )

"""
FelipedelosH
2026

Servicio: genera las combinaciones de contraseñas agrupadas por longitud.
- Retorna un iterador por cada longitud (chunk).
- Cada chunk sabe su total de combinaciones y su iterador de passwords.
"""
import itertools


class PasswordCombinationGenerator:
    @staticmethod
    def generateByLength(charset: str, min_len: int, max_len: int):
        """
        Generador de chunks. Cada yield entrega:
            (length, total_this_length, iterator_of_passwords)

        Ejemplo:
            for length, total, gen in BruteForceGenerator.generateByLength("abc", 1, 3):
                # chunk length=1 → 3 combinaciones
                # chunk length=2 → 9 combinaciones
                # chunk length=3 → 27 combinaciones
        """
        for length in range(min_len, max_len + 1):
            total = len(charset) ** length
            yield length, total, PasswordCombinationGenerator.generatorForLength(charset, length)

    @staticmethod
    def generatorForLength(charset: str, length: int):
        """Genera todas las combinaciones para UNA longitud específica."""
        for combo in itertools.product(charset, repeat=length):
            yield ''.join(combo)

    @staticmethod
    def countByLength(charset: str, length: int) -> int:
        return len(charset) ** length

    @staticmethod
    def countTotal(charset: str, min_len: int, max_len: int) -> int:
        return sum(len(charset) ** n for n in range(min_len, max_len + 1))

"""
FelipedelosH
2026

Servicio: colonia de termitas.
Cada termita es un .txt dentro de TEMP/TERMITES-L{length}/
El nombre del archivo es su índice (0.txt, 1.txt, ...)
El contenido del archivo es la password a probar.
"""
import os
import itertools


class TermiteService:
    @staticmethod
    def spawnColony(temp_path: str, charset: str, length: int, logger=None, start_index: int = 0) -> str:
        """
        Crea TEMP/TERMITES-L{length}/ y dentro un .txt por cada combinación.
        Si start_index > 0, solo genera las termitas desde start_index en adelante,
        conservando los nombres de índice absoluto (start_index.txt, start_index+1.txt, ...).
        Las termitas con índice < start_index (ya procesadas) se limpian de forma segura.
        Las termitas pendientes válidas que ya existen se reutilizan.
        Retorna la ruta de la carpeta creada.
        """
        if start_index < 0:
            if logger:
                logger.error(f"TermiteService::spawnColony::INVALID_START_INDEX::{start_index}")
            raise ValueError(f"start_index must be non-negative, got {start_index}")

        colony_path = os.path.join(temp_path, f"TERMITES-L{length}")
        os.makedirs(colony_path, exist_ok=True)

        total = len(charset) ** length

        # Clean up already-processed termites (index < start_index) safely.
        TermiteService._cleanupProcessedTermites(colony_path, start_index, logger)

        # Nothing left to generate: colony already exhausted up to total.
        if start_index >= total:
            if logger:
                logger.info(
                    f"TermiteService::spawnColony::LEN::{length}::START_INDEX::{start_index}::TOTAL::{total}::NOTHING_TO_GENERATE"
                )
            return colony_path

        # Generate only from start_index onwards. itertools.islice advances the
        # underlying product iterator without materializing skipped combinations.
        generated = 0
        reused = 0
        combinations = itertools.product(charset, repeat=length)
        for index, combo in enumerate(
            itertools.islice(combinations, start_index, total),
            start=start_index,
        ):
            password = ''.join(combo)
            file_path = os.path.join(colony_path, f"{index}.txt")
            if os.path.exists(file_path):
                existing = TermiteService.readTermite(file_path)
                if existing == password:
                    reused += 1
                    continue
                if logger:
                    logger.warn(
                        f"TermiteService::spawnColony::STALE_TERMITE::{index}.txt::REGENERATING"
                    )
            try:
                with open(file_path, 'w', encoding='utf-8') as f:
                    f.write(password)
            except Exception as e:
                if logger:
                    logger.error(f"TermiteService::spawnColony::ERR::{file_path}::{str(e)}")
            generated += 1

        if logger:
            logger.info(
                f"TermiteService::spawnColony::LEN::{length}::FROM_INDEX::{start_index}::GENERATED::{generated}::REUSED::{reused}"
            )

        return colony_path

    @staticmethod
    def _cleanupProcessedTermites(colony_path: str, start_index: int, logger=None):
        """Remove termite files with absolute index < start_index (already processed)."""
        if start_index <= 0:
            return
        try:
            entries = os.listdir(colony_path)
        except Exception as e:
            if logger:
                logger.warn(f"TermiteService::_cleanupProcessedTermites::ERR::{colony_path}::{str(e)}")
            return
        for entry in entries:
            if not entry.endswith('.txt'):
                continue
            try:
                index = int(entry.split('.')[0])
            except ValueError:
                continue
            if index < start_index:
                file_path = os.path.join(colony_path, entry)
                try:
                    os.remove(file_path)
                except Exception as e:
                    if logger:
                        logger.warn(f"TermiteService::_cleanupProcessedTermites::REMOVE_ERR::{file_path}::{str(e)}")

    @staticmethod
    def listColony(colony_path: str) -> list:
        """Lista los archivos de la colonia ordenados por índice numérico."""
        try:
            return sorted(
                os.listdir(colony_path),
                key=lambda x: int(x.split('.')[0])
            )
        except Exception:
            return []

    @staticmethod
    def readTermite(file_path: str):
        """Lee el contenido (password) de una termita."""
        try:
            with open(file_path, 'r', encoding='utf-8') as f:
                return f.read()
        except Exception:
            return None

    @staticmethod
    def kill(file_path: str):
        """La termita muere: se borra su archivo."""
        try:
            if os.path.exists(file_path):
                os.remove(file_path)
        except Exception:
            pass

    @staticmethod
    def survive(file_path: str, output_path: str):
        """La termita sobrevive: su archivo se mueve a OUTPUT."""
        try:
            filename = os.path.basename(file_path)
            dest = os.path.join(output_path, f"TERMITE_WINNER_{filename}")
            os.rename(file_path, dest)
        except Exception:
            pass

"""
FelipedelosH
2026

Servicio: operaciones sobre un archivo RAR.
- Probar contraseña
- Extraer contenido
"""
import os
import hashlib
import subprocess
from src.Models.Enums.PasswordTestResult import PasswordTestResult


class WinrarService:

    @staticmethod
    def tryPassword(rar_path: str, unrar_path: str, password: str, timeout: int = 10) -> PasswordTestResult:
        """
        Comprueba una contraseña contra el RAR.

        Retorna:
            - CORRECT  si la contraseña es correcta (returncode 0).
            - INCORRECT si la comprobación finalizó y la contraseña es incorrecta (returncode != 0).
            - ERROR    si ocurrió un fallo técnico transitorio (timeout u otro error de subprocess);
                        la comprobación NO finalizó y el progreso no debe avanzar.

        Los errores duros (FileNotFoundError, PermissionError) se relanzan: indican que el
        binario o el RAR desaparecieron y no se recuperan reintentando.
        """
        try:
            result = subprocess.run(
                [unrar_path, 't', '-p' + password, rar_path],
                stdout=subprocess.DEVNULL,
                stderr=subprocess.DEVNULL,
                timeout=timeout
            )
            if result.returncode == 0:
                return PasswordTestResult.CORRECT
            return PasswordTestResult.INCORRECT
        except FileNotFoundError:
            raise
        except PermissionError:
            raise
        except Exception:
            return PasswordTestResult.ERROR

    @staticmethod
    def computeSha256(path: str) -> str:
        """
        Calcula el SHA-256 del contenido de un archivo (lectura por chunks).
        Relanza FileNotFoundError si el archivo no existe.
        """
        hasher = hashlib.sha256()
        with open(path, "rb") as file:
            for chunk in iter(lambda: file.read(65536), b""):
                hasher.update(chunk)
        return hasher.hexdigest()

    @staticmethod
    def extractContent(rar_path: str, unrar_path: str, password: str, out_dir: str) -> str:
        """Extrae el RAR a la carpeta indicada y devuelve esa ruta."""
        os.makedirs(out_dir, exist_ok=True)

        result = subprocess.run(
            [unrar_path, 'x', '-p' + password, '-o+', rar_path, out_dir + os.sep],
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True
        )

        if result.returncode != 0:
            print("⚠️ No se pudo extraer:", result.stderr)
            return None

        return out_dir
